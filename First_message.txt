**Core Agent API (verbatim from latest SDK):**
```python
from browser_use import Agent, ChatBrowserUse, ChatOpenAI, ChatAnthropic
from dotenv import load_dotenv
import asyncio

load_dotenv()

async def main():
    agent = Agent(
        task="Natural language QA task here",
        llm=ChatOpenAI(model="gpt-4.1-mini"),          # or ChatAnthropic / ChatBrowserUse
        use_vision="auto",                               # "auto" / True / False
        max_failures=3,
        max_actions_per_step=3,
        generate_gif=False,
    )
    history = await agent.run(max_steps=100)

    # 🔑 AgentHistoryList methods — THESE ARE CRITICAL for RevGuard QA
    history.urls()                # List of visited URLs
    history.screenshot_paths()    # List of screenshot file paths
    history.screenshots()         # List of base64 screenshots
    history.action_names()        # Executed action names
    history.extracted_content()   # All extracted content
    history.errors()              # Errors per step
    history.final_result()        # Final extracted content
    history.is_done()             # True if completed successfully
    history.is_successful()       # True/None
    history.has_errors()          # Any errors occurred?
    history.model_thoughts()      # Agent reasoning process
    history.number_of_steps()
    history.total_duration_seconds()
```

### 3. Async Queue: Celery 5.6 + Redis + FastAPI
**Celery Docs:** https://docs.celeryq.dev/en/stable/getting-started/introduction.html
**FastAPI BackgroundTasks Caveat:** https://fastapi.tiangolo.com/tutorial/background-tasks/

> ⚠️ **CRITICAL VERDICT from FastAPI docs:** "If you need to perform heavy background computation… you might benefit from using other bigger tools like Celery." This confirms the architecture decision in ARCHITECTURE.MD.

**Production setup pattern (verified):**
```python
# celery_app.py
from celery import Celery
import os

celery_app = Celery(
    "worker",
    broker=os.getenv("REDIS_URL", "redis://localhost:6379/0"),
    backend=os.getenv("REDIS_URL", "redis://localhost:6379/1"),
)
celery_app.conf.update(
    task_track_started=True,           # enables STARTED state
    task_time_limit=180,               # hard kill at 180s (tests 30-120s)
    task_soft_time_limit=170,          # soft warning
    worker_prefetch_multiplier=1,      # fairness for long QA tasks
    task_acks_late=True,               # prevent lost tasks on worker crash
    task_reject_on_worker_lost=True,   # requeue on worker crash
)

# worker.py
from celery_app import celery_app

@celery_app.task(bind=True, max_retries=3, autoretry_for=(Exception,), retry_backoff=True)
def run_browser_test(self, job_id: str, prompt: str) -> dict:
    """Run a browser-use QA test in the background."""
    # Update progress (frontend polls this)
    self.update_state(state="PROGRESS", meta={"stage": "initializing"})
    # ... run Agent here ...
    # On exception: self.retry(exc=exc, countdown=2**self.request.retries)
    return {"job_id": job_id, "result": "...", "screenshots": [...]}

# main.py — FastAPI endpoints
from fastapi import FastAPI, HTTPException
from celery.result import AsyncResult
from pydantic import BaseModel

app = FastAPI()

class TestRunRequest(BaseModel):
    prompt: str

@app.post("/api/tests/run")
async def enqueue_test(req: TestRunRequest):
    job_id = str(uuid.uuid4())
    task = run_browser_test.delay(job_id, req.prompt)
    return {"job_id": job_id, "task_id": task.id}

@app.get("/api/tests/status/{task_id}")
def get_status(task_id: str):
    result = AsyncResult(task_id, app=celery_app)
    # Celery states: PENDING → STARTED → PROGRESS → SUCCESS / FAILURE
    return {
        "task_id": task_id,
        "state": result.state,
        "progress": result.info if result.state == "PROGRESS" else None,
        "result": result.result if result.state == "SUCCESS" else None,
        "error": str(result.info) if result.state == "FAILURE" else None,
    }
```


### 6. Enterprise Integrations (Moat)

#### Linear GraphQL — Auto-Ticketing
**Docs:** https://developers.linear.app/docs/graphql/working-with-the-graphql-api
```python
import requests
import os

LINEAR_API_KEY = os.environ["LINEAR_API_KEY"]
GRAPHQL_ENDPOINT = "https://api.linear.app/graphql"
TEAM_ID = os.environ["LINEAR_TEAM_ID"]  # get from: query { teams { nodes { id name }}}

def create_failure_issue(title: str, description_md: str) -> dict:
    mutation = """
    mutation IssueCreate($input: IssueCreateInput!) {
        issueCreate(input: $input) {
            success
            issue { id title identifier }
        }
    }
    """
    variables = {
        "input": {
            "title": title,
            "description": description_md,  # markdown supported
            "teamId": TEAM_ID,
        }
    }
    resp = requests.post(
        GRAPHQL_ENDPOINT,
        headers={
            "Content-Type": "application/json",
            "Authorization": f"Bearer {LINEAR_API_KEY}",  # OAuth2: Bearer <token>; Personal key: bare <API_KEY>
        },
        json={"query": mutation, "variables": variables},
    )
    return resp.json()
```
> Note: Linear images require auth — download failure screenshots and self-host before embedding tickets.

#### Resend Python SDK — Email Alerts
**Docs:** https://resend.com/docs/send-with-python
```python
# pip install resend
import os
import resend

resend.api_key = os.environ["RESEND_API_KEY"]

def send_failure_alert(to_email: str, test_name: str, screenshot_path: str, steps_summary: str):
    # Read screenshot as attachment
    with open(screenshot_path, "rb") as f:
        attachment_data = f.read()
    
    params: resend.Emails.SendParams = {
        "from": "Leaka AI <qa@leaka.ai>",
        "to": [to_email],
        "subject": f"🚨 Test FAILED: {test_name}",
        "html": f"""
        <h2>Test Failure Alert</h2>
        <p><strong>Test:</strong> {test_name}</p>
        <h3>Steps before failure:</h3>
        <pre>{steps_summary}</pre>
        <p>Full screenshot attached.</p>
        """,
        "reply_to": "support@leaka.ai",
        "attachments": [
            {"filename": "failure_screenshot.png", "content": list(attachment_data)}
        ],
        "tags": [{"name": "category", "value": "qa-failure"}],
    }
    return resend.Emails.send(params)
```
